# 计算策略收益（无成本）

df['returns'] = df['close'].pct_change()
df['strategy_returns'] = df['signal'].shift(1) * df['returns']
df['cumulative_strategy'] = (1 + df['strategy_returns']).fillna(1).cumprod()


df['returns'] = df['close'].pct_change()
strategy_returns = df['strategy_returns'].dropna()


# 年化收益率
total_return = df['cumulative_strategy'].iloc[-1] - 1
years = len(df) / 252
annual_return = (1 + total_return) ** (1 / years) - 1

# 夏普比率（无风险利率2%）
risk_free = 0.02
excess_return = strategy_returns.mean() * 252 - risk_free
sharpe = excess_return / (strategy_returns.std() * (252**0.5))


# 最大回撤
cumulative = (1 + strategy_returns).cumprod()
peak = cumulative.expanding().max()
max_drawdown = (cumulative / peak - 1).min()


# 胜率
win_rate = (strategy_returns > 0).sum() / (strategy_returns != 0).sum()


print(f"年化收益率: {annual_return:.2%}")
print(f"夏普比率: {sharpe:.2f}")
print(f"最大回撤: {max_drawdown:.2%}")
print(f"胜率: {win_rate:.2%}")
